#include <stdio.h>

void FIFO(int pages[], int n, int capacity) {
    int frames[capacity];  // To store pages in memory
    int front = 0;         // Points to the page to be replaced
    int count = 0;         // Number of pages currently in frames
    int pageFaults = 0;

    for (int i = 0; i < capacity; i++) {
        frames[i] = -1;    // Initialize frames as empty
    }

    for (int i = 0; i < n; i++) {
        int page = pages[i];
        int found = 0;

        // Check if the page is already in frames
        for (int j = 0; j < count; j++) {
            if (frames[j] == page) {
                found = 1; // Page found, no page fault
                break;
            }
        }

        if (!found) { // Page fault occurred
            if (count < capacity) {
                frames[count] = page; // Add page if there's space
                count++;
            } else {
                frames[front] = page; // Replace the oldest page
                front = (front + 1) % capacity; // Update the pointer
            }
            pageFaults++;
        }

        // Print the current state of frames
        printf("Frames after accessing page %d: ", page);
        for (int j = 0; j < capacity; j++) {
            if (frames[j] != -1)
                printf("%d ", frames[j]);
            else
                printf("_ ");
        }
        printf("\n");
    }

    printf("\nTotal Page Faults: %d\n", pageFaults);
}

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2, 1, 2};
    int n = sizeof(pages) / sizeof(pages[0]);
    int capacity = 3;

    printf("FIFO Page Replacement Algorithm:\n");
    FIFO(pages, n, capacity);

    return 0;
}
