#include <iostream>
#include <queue>
#include <thread>
#include <semaphore.h>
#include <chrono>
#include <random>

using namespace std;


queue<int> buffer;
const int buffer_size = 5;  

sem_t empty_slots; 
sem_t full_slots;   
sem_t buffer_access;  


void producer() {
    while (true) {
        
        int item = rand() % 100 + 1; 

      
        sem_wait(&empty_slots);

        sem_wait(&buffer_access);
        buffer.push(item);  
        cout << "Produced: " << item << " | Buffer size: " << buffer.size() << endl;
        sem_post(&buffer_access);

    
        sem_post(&full_slots);

     
        this_thread::sleep_for(chrono::seconds(rand() % 3 + 1));
    }
}


void consumer() {
    while (true) {

        sem_wait(&full_slots);

    
        sem_wait(&buffer_access);
        int item = buffer.front();  
        buffer.pop();
        cout << "Consumed: " << item << " | Buffer size: " << buffer.size() << endl;
        sem_post(&buffer_access);

        sem_post(&empty_slots);

    
        this_thread::sleep_for(chrono::seconds(rand() % 3 + 1));
    }
}

int main() {
 
    sem_init(&empty_slots, 0, buffer_size);  
    sem_init(&full_slots, 0, 0);             
    sem_init(&buffer_access, 0, 1);        

    srand(time(0));

    thread producer_thread(producer);
    thread consumer_thread(consumer);

  
    producer_thread.join();
    consumer_thread.join();

    sem_destroy(&empty_slots);
    sem_destroy(&full_slots);
    sem_destroy(&buffer_access);

    return 0;
}
