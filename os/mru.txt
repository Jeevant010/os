#include <stdio.h>

void MRU(int pages[], int n, int capacity) {
    int frames[capacity];
    int used[capacity];
    int count = 0, pageFaults = 0;

    for (int i = 0; i < capacity; i++) {
        frames[i] = -1; // Initialize frames as empty
        used[i] = 0;
    }

    for (int i = 0; i < n; i++) {
        int page = pages[i];
        int found = 0;

        // Check if page is already in frames
        for (int j = 0; j < count; j++) {
            if (frames[j] == page) {
                used[j] = i; // Update most recently used
                found = 1;
                break;
            }
        }

        // Page not found in frames (page fault)
        if (!found) {
            if (count < capacity) {
                frames[count] = page;
                used[count] = i;
                count++;
            } else {
                // Find the most recently used page to replace
                int mruIndex = 0;
                for (int j = 1; j < capacity; j++) {
                    if (used[j] > used[mruIndex]) {
                        mruIndex = j;
                    }
                }
                frames[mruIndex] = page;
                used[mruIndex] = i;
            }
            pageFaults++;
        }

        // Print current frame status
        printf("Frames: ");
        for (int j = 0; j < capacity; j++) {
            if (frames[j] != -1)
                printf("%d ", frames[j]);
            else
                printf("_ ");
        }
        printf("\n");
    }

    printf("Total Page Faults: %d\n", pageFaults);
}

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2, 1, 2};
    int n = sizeof(pages) / sizeof(pages[0]);
    int capacity = 3;

    printf("MRU Algorithm:\n");
    MRU(pages, n, capacity);

    return 0;
}
