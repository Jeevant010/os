#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>

#define NUM_PHILOSOPHERS 5

sem_t forks[NUM_PHILOSOPHERS];  // Semaphores to represent forks
pthread_t philosophers[NUM_PHILOSOPHERS];

// Function to simulate philosopher's thinking and eating
void* philosopher(void* num) {
    int philosopher_id = *((int*)num);

    while (1) {
        printf("Philosopher %d is thinking.\n", philosopher_id);
        // Simulate thinking by sleeping for a random time
        sleep(rand() % 3);

        // Trying to pick up both forks
        printf("Philosopher %d is hungry.\n", philosopher_id);
        
        // Picking up left fork
        sem_wait(&forks[philosopher_id]);
        printf("Philosopher %d picked up left fork.\n", philosopher_id);

        // Picking up right fork
        sem_wait(&forks[(philosopher_id + 1) % NUM_PHILOSOPHERS]);
        printf("Philosopher %d picked up right fork.\n", philosopher_id);

        // Eating
        printf("Philosopher %d is eating.\n", philosopher_id);
        sleep(rand() % 3);  // Simulate eating for a random time

        // Putting down forks
        sem_post(&forks[philosopher_id]);
        printf("Philosopher %d put down left fork.\n", philosopher_id);

        sem_post(&forks[(philosopher_id + 1) % NUM_PHILOSOPHERS]);
        printf("Philosopher %d put down right fork.\n", philosopher_id);
    }

    return NULL;
}

int main() {
    // Initialize semaphores for each fork
    for (int i = 0; i < NUM_PHILOSOPHERS; i++) {
        sem_init(&forks[i], 0, 1);  // Each fork is available at the beginning
    }

    // Create philosopher threads
    int philosopher_ids[NUM_PHILOSOPHERS];
    for (int i = 0; i < NUM_PHILOSOPHERS; i++) {
        philosopher_ids[i] = i;
        pthread_create(&philosophers[i], NULL, philosopher, &philosopher_ids[i]);
    }

    // Wait for all philosopher threads to finish (which they never will in this case)
    for (int i = 0; i < NUM_PHILOSOPHERS; i++) {
        pthread_join(philosophers[i], NULL);
    }

    // Destroy semaphores
    for (int i = 0; i < NUM_PHILOSOPHERS; i++) {
        sem_destroy(&forks[i]);
    }

    return 0;
}
