1. Shell script Program to
accept a character and check whether it is an
(i)
Lower case alphabet
(ii)
Upper case alphabet
(iii)
A digit
(iv)
Special symbol
(v)
Vowel
#!/bin/bash
echo "Enter a character: "
read char

case $char in
    [a-z])
        echo "$char is a lowercase alphabet."
        ;;
    [A-Z])
        echo "$char is an uppercase alphabet."
        ;;
    [0-9])
        echo "$char is a digit."
        ;;
    [aeiouAEIOU])
        echo "$char is a vowel."
        ;;
    *)
        echo "$char is a special symbol."
        ;;
esac

3. Shell Script Program to check whether given file is a directory or
not.

#!/bin/bash
echo "Enter the file/directory name: "
read name

if [ -d "$name" ]; then
    echo "$name is a directory."
else
    echo "$name is not a directory."
fi

4. Shell Script Program to count number of files in a Directory.

#!/bin/bash
echo "Enter the directory name: "
read dir

if [ -d "$dir" ]; then
    count=$(find "$dir" -type f | wc -l)
    echo "Number of files in $dir: $count"
else
    echo "$dir is not a directory."
fi



5.Shell Script Program to copy contents of one file to another.

#!/bin/bash
echo "Enter the source file name: "
read src
echo "Enter the destination file name: "
read dest

if [ -f "$src" ]; then
    cp "$src" "$dest"
    echo "Contents of $src have been copied to $dest."
else
    echo "Source file does not exist."
fi



6. Add two numbers supplied as command line arguments

#!/bin/bash
a=$1
b=$2
read a;
read b;
let sum=$a+$b
echo "The sum of $a and $b is $sum"




7. Find the biggest of three numbers supplied as command line arguments

#!/bin/bash
if [ $# -ne 3 ]; then
    echo "Usage: $0 num1 num2 num3"
    exit 1
fi

if [ $1 -ge $2 ] && [ $1 -ge $3 ]; then
    echo "The largest number is $1."
elif [ $2 -ge $1 ] && [ $2 -ge $3 ]; then
    echo "The largest number is $2."
else
    echo "The largest number is $3."
fi
