#include <stdio.h>

void round_robin(int n, int burst[], int arrival[], int quantum) {
    int remaining[n], wait[n], tat[n], total_wait = 0, total_tat = 0;
    for (int i = 0; i < n; i++) remaining[i] = burst[i];

    int current_time = 0;
    int completed_count = 0;

    while (completed_count < n) {
        for (int i = 0; i < n; i++) {
            if (arrival[i] <= current_time && remaining[i] > 0) {
                if (remaining[i] <= quantum) {
                    current_time += remaining[i];
                    tat[i] = current_time - arrival[i];
                    wait[i] = tat[i] - burst[i];
                    remaining[i] = 0;
                    completed_count++;
                    total_wait += wait[i];
                    total_tat += tat[i];
                } else {
                    current_time += quantum;
                    remaining[i] -= quantum;
                }
            }
        }
    }

    printf("Process\tArrival\tBurst\tWaiting\tTurn-Around\n");
    for (int i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\n", i + 1, arrival[i], burst[i], wait[i], tat[i]);
    }
    printf("Average Waiting Time: %.2f\n", (float)total_wait / n);
    printf("Average Turn-Around Time: %.2f\n", (float)total_tat / n);
}

int main() {
    int n = 4;
    int burst[] = {8, 5, 9, 6};
    int arrival[] = {0, 1, 2, 3};
    int quantum = 4;

    printf("Round Robin Scheduling\n");
    round_robin(n, burst, arrival, quantum);

    return 0;
}
