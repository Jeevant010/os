#include <stdio.h>

void priority(int n, int burst[], int arrival[], int priority[]) {
    int wait[n], tat[n], total_wait = 0, total_tat = 0;
    int completed[n];
    for (int i = 0; i < n; i++) completed[i] = 0;

    int current_time = 0, completed_count = 0;

    while (completed_count < n) {
        int highest_priority = -1, min_priority = 1e9;
        for (int i = 0; i < n; i++) {
            if (!completed[i] && arrival[i] <= current_time && priority[i] < min_priority) {
                min_priority = priority[i];
                highest_priority = i;
            }
        }

        if (highest_priority == -1) {
            current_time++;
            continue;
        }

        wait[highest_priority] = current_time - arrival[highest_priority];
        tat[highest_priority] = wait[highest_priority] + burst[highest_priority];
        current_time += burst[highest_priority];
        completed[highest_priority] = 1;
        completed_count++;
        total_wait += wait[highest_priority];
        total_tat += tat[highest_priority];
    }

    printf("Process\tArrival\tBurst\tPriority\tWaiting\tTurn-Around\n");
    for (int i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t\t%d\t%d\n", i + 1, arrival[i], burst[i], priority[i], wait[i], tat[i]);
    }
    printf("Average Waiting Time: %.2f\n", (float)total_wait / n);
    printf("Average Turn-Around Time: %.2f\n", (float)total_tat / n);
}

int main() {
    int n = 4;
    int burst[] = {6, 8, 7, 3};
    int arrival[] = {0, 1, 2, 3};
    int priority[] = {2, 1, 3, 2};

    printf("Priority Scheduling\n");
    priority(n, burst, arrival, priority);

    return 0;
}
